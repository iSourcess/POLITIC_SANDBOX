-- Ejecuta este código en el SQL Editor de tu panel de Supabase
-- IMPORTANTE: Si la consola te da un error que diga algo sobre "bigint" y "uuid", 
-- debes cambiar el tipo de dato de "poll_id" de "uuid" a "bigint" antes de ejecutarlo.

CREATE TABLE public.poll_reactions (
  id bigint GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  poll_id uuid NOT NULL,
  user_id uuid NOT NULL,
  reaction_type text NOT NULL,
  created_at timestamp with time zone default timezone('utc'::text, now()),
  UNIQUE(poll_id, user_id)
);

-- Deshabilitamos temporalmente RLS (Row Level Security) para asegurar que se pueda 
-- escribir/leer libremente desde la aplicación. Puedes configurar las políticas más tarde.
ALTER TABLE public.poll_reactions DISABLE ROW LEVEL SECURITY;
